¿Cuáles direcciones puedes usar dentro de tu red?

Privadas, públicas y especiales

import ipaddress as ipa
BLOQUES = [
    ("10.0.0.0/8", "privada"),
    ("172.16.0.0/12", "privada"),
    ("192.168.0.0/16", "privada"),
    ("127.0.0.0/8", "loopback"),
    ("169.254.0.0/16", "enlace local"),
]
def clasificar(texto):
    try:
        ip = ipa.ip_address(texto)
    except ValueError:
        return "inválida"
    for red, tipo in BLOQUES:
        if ip in ipa.ip_network(red):
            return tipo
    return "pública"
for d in direcciones:
    print(d, clasificar(d))

¿Y si cada grupo necesita un tamaño distinto?

VLSM: cada grupo con su tamaño

import ipaddress
grupos = {"enlace": 2, "sensores": 60,
          "servidores": 10, "administracion": 25}
orden = sorted(grupos.items(), key=lambda p: -p[1])
inicio = int(ipaddress.ip_address("192.168.10.0"))
for nombre, equipos in orden:
    bits = 2
    while 2 ** bits < equipos + 2:
        bits += 1
    red = ipaddress.ip_network((inicio, 32 - bits))
    print(nombre, red, red.num_addresses - 2)
    inicio += 2 ** bits
print(ipaddress.ip_address(inicio))
