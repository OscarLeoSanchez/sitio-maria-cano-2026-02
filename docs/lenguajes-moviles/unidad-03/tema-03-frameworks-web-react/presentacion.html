useState: la memoria del componente

import { useState } from 'react';

function Contador() {
  const [conteo, setConteo] = useState(0);

  function incrementar() {
    setConteo(conteo + 1);
  }

  return (
    <div>
      <p>Has presionado el botón {conteo} veces</p>
      <button onClick={incrementar}>Incrementar</button>
    </div>
  );
}

setConteo actualiza el valor y le avisa a React que debe reconstruir el componente, el mismo papel que setState cumple en un StatefulWidget de Flutter.
