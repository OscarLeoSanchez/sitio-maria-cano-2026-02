¿Cuántos pedidos hay que mirar para encontrar uno?

Buscar es bajar por un camino

def buscar(self, dato):
    return self._buscar(self.raiz, dato)

def _buscar(self, nodo, dato):
    if nodo is None:
        return False  # caso base: se acabo el arbol y no aparecio
    if dato == nodo.dato:
        return True  # caso base: lo encontro
    return self._buscar(nodo.izquierdo, dato) if dato < nodo.dato else self._buscar(nodo.derecho, dato)
def insertar(self, dato):
    self.raiz = self._insertar(self.raiz, dato)

def _insertar(self, nodo, dato):
    if nodo is None:
        return NodoArbol(dato)  # caso base: aqui va el nodo nuevo
    if dato < nodo.dato:
        nodo.izquierdo = self._insertar(nodo.izquierdo, dato)
    elif dato > nodo.dato:
        nodo.derecho = self._insertar(nodo.derecho, dato)
    # dato == nodo.dato: valor repetido, no se hace nada
    return nodo
El subárbol descartado nunca se visita.

¿Cómo listar los pedidos de menor a mayor?

Listar en orden sin ordenar

def inorden(self):
    resultado = []
    self._inorden(self.raiz, resultado)
    return resultado

def _inorden(self, nodo, resultado):
    if nodo is None:
        return  # caso base: subarbol vacio, nada que visitar
    self._inorden(nodo.izquierdo, resultado)
    resultado.append(nodo.dato)
    self._inorden(nodo.derecho, resultado)
def rango(self, desde, hasta):
    resultado = []
    self._rango(self.raiz, desde, hasta, resultado)
    return resultado

def _rango(self, nodo, desde, hasta, resultado):
    if nodo is None:
        return
    if desde < nodo.dato:
        self._rango(nodo.izquierdo, desde, hasta, resultado)  # puede haber pedidos del rango a la izquierda
    if desde <= nodo.dato <= hasta:
        resultado.append(nodo.dato)
    if nodo.dato < hasta:
        self._rango(nodo.derecho, desde, hasta, resultado)  # puede haber pedidos del rango a la derecha

¿Qué pasa con los hijos cuando se cancela un nodo?

Cancelar depende de cuántos hijos tiene

def eliminar(self, dato):
    self.raiz = self._eliminar(self.raiz, dato)

def _eliminar(self, nodo, dato):
    if nodo is None:
        return None  # el dato no estaba en el arbol
    if dato < nodo.dato:
        nodo.izquierdo = self._eliminar(nodo.izquierdo, dato)
    elif dato > nodo.dato:
        nodo.derecho = self._eliminar(nodo.derecho, dato)
    elif nodo.izquierdo is not None and nodo.derecho is not None:
        # dos hijos: copia el sucesor inorden y elimina ese sucesor de su lugar
        nodo.dato = self._minimo(nodo.derecho)
        nodo.derecho = self._eliminar(nodo.derecho, nodo.dato)
    else:
        # hoja o un solo hijo: el hijo que exista, o None, toma el lugar
        nodo = nodo.izquierdo if nodo.izquierdo is not None else nodo.derecho
    return nodo
def _minimo(self, nodo):
    while nodo.izquierdo is not None:
        nodo = nodo.izquierdo
    return nodo.dato

¿Cómo sabe el árbol que se está torciendo?

Un número mide cuánto se tuerce cada nodo

class NodoAVL:
    def __init__(self, dato):
        self.dato = dato
        self.izquierdo = None
        self.derecho = None
        self.altura = 0
class ArbolAVL:
    def __init__(self):
        self.raiz = None

    def _altura(self, nodo):
        return nodo.altura if nodo is not None else -1

    def _factor(self, nodo):
        return self._altura(nodo.derecho) - self._altura(nodo.izquierdo)

    def _actualizar_altura(self, nodo):
        nodo.altura = 1 + max(self._altura(nodo.izquierdo), self._altura(nodo.derecho))
Con la altura guardada en cada nodo, el factor sale sin recorrer subárboles.
En un árbol AVL todo factor vale -1, 0 o 1.

¿Cómo se endereza un árbol que se tuerce hacia un lado?

Una rotación endereza tres nodos

def _rotar_izquierda(self, nodo):
    nueva_raiz = nodo.derecho
    nodo.derecho = nueva_raiz.izquierdo
    nueva_raiz.izquierdo = nodo
    self._actualizar_altura(nodo)
    self._actualizar_altura(nueva_raiz)
    return nueva_raiz
def _rotar_derecha(self, nodo):
    nueva_raiz = nodo.izquierdo
    nodo.izquierdo = nueva_raiz.derecho
    nueva_raiz.derecho = nodo
    self._actualizar_altura(nodo)
    self._actualizar_altura(nueva_raiz)
    return nueva_raiz
Factor +2: _rotar_izquierda
Factor -2: _rotar_derecha

¿Y si el árbol se tuerce en zigzag?

Dos rotaciones enderezan el zigzag

def _balancear(self, nodo):
    self._actualizar_altura(nodo)
    factor = self._factor(nodo)
    if factor < -1:
        if self._factor(nodo.izquierdo) > 0:
            nodo.izquierdo = self._rotar_izquierda(nodo.izquierdo)  # LR
        return self._rotar_derecha(nodo)
    if factor > 1:
        if self._factor(nodo.derecho) < 0:
            nodo.derecho = self._rotar_derecha(nodo.derecho)  # RL
        return self._rotar_izquierda(nodo)
    return nodo  # ya esta dentro de {-1, 0, 1}, nada que hacer
NodoHijoCasoRotaciones
-2-1 o 0LL, cadena a la izquierda_rotar_derecha
-2+1LR, zigzag_rotar_izquierda al hijo y _rotar_derecha
+2+1 o 0RR, cadena a la derecha_rotar_izquierda
+2-1RL, zigzag_rotar_derecha al hijo y _rotar_izquierda
Una rotación sola no basta si el hijo se tuerce hacia el otro lado.

def _insertar(self, nodo, dato):
    if nodo is None:
        return NodoAVL(dato)  # caso base: aqui va el nodo nuevo
    if dato < nodo.dato:
        nodo.izquierdo = self._insertar(nodo.izquierdo, dato)
    elif dato > nodo.dato:
        nodo.derecho = self._insertar(nodo.derecho, dato)
    else:
        return nodo  # valor repetido, no se hace nada
    return self._balancear(nodo)  # al volver, corrige
