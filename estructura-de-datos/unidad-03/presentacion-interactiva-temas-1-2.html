class Cola:

            self._cabeza = nuevo
        else:
            self._ultimo.siguiente = nuevo

    def desencolar(self):
        eliminado = self._cabeza
        self._cabeza = eliminado.siguiente
        eliminado.siguiente = None
        return eliminado.dato

LIFO sobre una lista

Pila: todo ocurre en la cabeza

class Pila:
    def __init__(self):
        self._cabeza = None

    def apilar(self, elemento):
        nuevo = Nodo(elemento)
        nuevo.siguiente = self._cabeza
        self._cabeza = nuevo
LIFO: sale primero quien llegó último. Sin recorridos: costo constante.

Todo junto: el reproductor

class Reproductor:
    def __init__(self):
        self.a_continuacion = Cola()
        self.historial = Pila()
        self.sonando = None

    def agregar(self, cancion):
        self.a_continuacion.encolar(cancion)

    def siguiente(self):
        if self.sonando is not None:
            self.historial.apilar(self.sonando)
        self.sonando = self.a_continuacion.desencolar()

    def volver(self):
        self.sonando = self.historial.desapilar()
